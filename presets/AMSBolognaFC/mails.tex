\begin{frame}[c,fragile]{Mail Addresses}
%
\begin{block}{\texttt{\textbackslash{}umail} command\textemdash{}e.g., \umail{andrea.omicini@unibo.it}}
\begin{itemize}
	\item to be used for every mail address, the title page's included, instead of \texttt{\textbackslash{}url\{mailto:\ldots\}} or \texttt{\textbackslash{}texttt}
	\item[e.g.] \verb|\umail{andrea.omicini@unibo.it}| prints as \umail{andrea.omicini@unibo.it}
	\item[] linked to \texttt{mailto:}, printing the address only, as \texttt{\textbackslash{}uurl} prints a URL
\end{itemize}
\end{block}
%
\begin{block}{\texttt{\textbackslash{}uumail} command\textemdash{}e.g., \uumail{andrea.omicini@unibo.it}}
\begin{itemize}
	\item the same, smaller and quieter
	\item[e.g.] \verb|\uumail{andrea.omicini@unibo.it}| prints as \uumail{andrea.omicini@unibo.it}
\end{itemize}
\end{block}
%
\end{frame}
