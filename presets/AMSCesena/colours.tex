\begin{frame}[c,fragile]{Colours}
%
The palette is \alert{CesenaSMA2006}'s, and every colour takes the role its
counterpart has in \alert{AMSUniBo}: \textcolor{owblue}{\textit{blue}} identifies,
\textcolor{owbluedeep}{\textit{deep blue}} structures, while alerts stay Beamer's \alert{red}
%
\begin{exampleblock}{The four that carry the style}
\begin{tabular}{@{}llll@{}}
\swatch{owblue}     & \texttt{owblue}     & \texttt{\#3333B2} & title box, author, bullets, links \\
\swatch{owbluedeep} & \texttt{owbluedeep} & \texttt{\#262686} & block headers, emphasis, \textsc{toc} \\
\swatch{owblack}    & \texttt{owblack}    & \texttt{\#000000} & head line, author in the foot line \\
\swatch{owred}      & \texttt{owred}      & \texttt{\#FF0000} & alerts \\
\end{tabular}
\end{exampleblock}
%
\begin{itemize}
	\item \texttt{owblue} is Beamer's own default blue, which \alert{CesenaSMA2006} kept
	\item \texttt{owbluedeep}, the black and the block fills are what \texttt{whale} and
	      \texttt{orchid} made of it
\end{itemize}
%
\end{frame}
