\begin{frame}[c,fragile]{Colours}
%
The palette is the old \alert{AMSCesenaPurpleAndGold}'s, and every colour takes the
role its counterpart has in \alert{AMSUniBo}: \textcolor{pggolddark}{\textit{gold}} identifies,
\textcolor{pgpurple}{\textit{purple}} structures
%
\begin{exampleblock}{The four that carry the style}
\begin{tabular}{@{}llll@{}}
\swatch{pggold}     & \texttt{pggold}     & \texttt{\#D4AF38} & head line \\
\swatch{pggolddark} & \texttt{pggolddark} & \texttt{\#7F6922} & author, bullets, alerts, links \\
\swatch{pgpurple}   & \texttt{pgpurple}   & \texttt{\#660066} & block headers, emphasis, \textsc{toc} \\
\swatch{pggoldpale} & \texttt{pggoldpale} & \texttt{\#EDE8AB} & over purple: header titles \\
\end{tabular}
\end{exampleblock}
%
\begin{itemize}
	\item \texttt{pggold}, \texttt{pggoldpale} and \texttt{pgpurple} are \emph{Gold (metallic)},
	      \emph{Pale Goldenrod} and \emph{Cesena Purple}, as \alert{AMSCesenaPurpleAndGold} defined them
	\item \texttt{pggolddark} is the gold darkened to read on white, which the old style's never did
\end{itemize}
%
\end{frame}
