\begin{frame}[c,fragile]{Colours}
%
The palette is the old \alert{AMSCesena}'s, and every colour takes the role its
counterpart has in \alert{AMSUniBo}: \textcolor{rgred}{\textit{red}} identifies,
\textcolor{rgreddeep}{\textit{deep red}} and \textcolor{rggreymid}{\textit{grey}} structure
%
\begin{exampleblock}{The four that carry the style}
\begin{tabular}{@{}llll@{}}
\swatch{rgred}     & \texttt{rgred}     & \texttt{\#CC001A} & head line, author, alerts, links \\
\swatch{rgreddeep} & \texttt{rgreddeep} & \texttt{\#850011} & block headers, emphasis, \textsc{toc} \\
\swatch{rgsilver}   & \texttt{rgsilver}   & \texttt{\#B3B3B3} & separation lines, and the bands and fills \\
\swatch{rggreymid}  & \texttt{rggreymid}  & \texttt{\#5A5A5A} & subtitle, date, footnotes \\
\end{tabular}
\end{exampleblock}
%
\begin{itemize}
	\item \texttt{rgred} is the \texttt{darkred} \alert{AMSCesena} defined;
	      \texttt{rgreddeep} is the shade it gave its block headers, \texttt{rgreddark} its bullets
	\item \texttt{rgsilver} and its tints are the greys of Beamer's \texttt{beaver}, which
	      \alert{AMSCesena} was built on
\end{itemize}
%
\end{frame}
